D. Redirects and canonical domain rules

Force https

Force your single domain form either with or without www

Example Express middleware

app.use((req, res, next) => {
  if (req.headers["x-forwarded-proto"] !== "https") {
    return res.redirect(301, "https://" + req.headers.host + req.url);
  }
  next();
});
app.use((req, res, next) => {
  const host = req.headers.host;
  if (host === "www.kiminouknox.com") {
    return res.redirect(301, "https://kiminouknox.com" + req.url);
  }
  next();
});
