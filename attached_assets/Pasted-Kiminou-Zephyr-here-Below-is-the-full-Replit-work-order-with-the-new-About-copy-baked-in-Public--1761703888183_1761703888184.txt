### 8b) Content sweep script

Only touch user facing content files such as markdown CMS JSON or HTML in `/content` or `/pages`. Do not edit CSS JS or config.

```js
// scripts/clean-dashes.js
import fs from "fs";
import path from "path";

const ROOT = path.resolve("content"); // adjust if your content folder differs

function walk(dir){
  for(const f of fs.readdirSync(dir)){
    const p = path.join(dir, f);
    const s = fs.statSync(p);
    if(s.isDirectory()) walk(p);
    else if(/\.(md|mdx|html|txt|json)$/i.test(f)){
      let t = fs.readFileSync(p, "utf8");
      // replace visible dash like characters with a space
      t = t.replace(/[\u002d\u00ad\u2010-\u2015]/g, " ");
      fs.writeFileSync(p, t, "utf8");
    }
  }
}
walk(ROOT);
console.log("Dash cleanup complete");
```

Run: `node scripts/clean-dashes.js`
